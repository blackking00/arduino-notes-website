Example 1 LED Blink — Pehla Program! 
// Pin 13 par LED lagaao
int ledPin = 13;

void setup() {
  pinMode(ledPin, OUTPUT); // ledPin ko output banao
}

void loop() {
  digitalWrite(ledPin, HIGH); // LED ON
  delay(500);                  // 0.5 second ruko
  digitalWrite(ledPin, LOW);  // LED OFF
  delay(500);                  // 0.5 second ruko
}
Example 3 Button se LED Control 
int buttonPin = 2;   // Button pin 2 par
int ledPin    = 13;  // LED pin 13 par
int buttonState;

void setup() {
  pinMode(buttonPin, INPUT);
  pinMode(ledPin, OUTPUT);
}

void loop() {
  buttonState = digitalRead(buttonPin);

  if (buttonState == HIGH) {
    digitalWrite(ledPin, HIGH); // LED ON
  } else {
    digitalWrite(ledPin, LOW);  // LED OFF
  }
}
Example 5 Potentiometer se LED Brightness 
int potPin    = A0;  // Potentiometer A0 par
int ledPin    = 9;   // LED PWM pin par
int potValue;
int brightness;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  potValue   = analogRead(potPin);   // 0 se 1023
  brightness = potValue / 4;          // 0 se 255
  analogWrite(ledPin, brightness);    // Brightness set
  delay(100);
}
Example 15 millis() — Non-Blocking Blink 
int  ledPin        = 13;
bool ledState      = false;
long previousMillis = 0;
long interval       = 1000;

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  long currentMillis = millis();

  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;
    ledState = !ledState;           // Toggle
    digitalWrite(ledPin, ledState);
  }
  // Yahan aur kaam bhi kar sakte hain!
}
Example 19 Servo Motor Sweep 
#include <Servo.h>

Servo myServo;

void setup() {
  myServo.attach(9); // Pin 9 se attach
}

void loop() {
  // 0 se 180 degree tak sweep
  for (int a = 0; a <= 180; a += 10) {
    myServo.write(a);
    delay(100);
  }
  // 180 se 0 tak wapas
  for (int a = 180; a >= 0; a -= 10) {
    myServo.write(a);
    delay(100);
  }
}
